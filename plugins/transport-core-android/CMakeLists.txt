cmake_minimum_required(VERSION 3.22.1)
project(transportnative)
set(CMAKE_CXX_STANDARD 17)
add_library(transportnative SHARED src/main/cpp/TransportJni.cpp)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(RUST_TRIPLE aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
  set(RUST_TRIPLE armv7-linux-androideabi)
elseif(ANDROID_ABI STREQUAL "x86")
  set(RUST_TRIPLE i686-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(RUST_TRIPLE x86_64-linux-android)
else()
  message(FATAL_ERROR "Unsupported transport ABI: ${ANDROID_ABI}")
endif()
add_library(transport_rust STATIC IMPORTED)
set_target_properties(transport_rust PROPERTIES IMPORTED_LOCATION
  ${CMAKE_CURRENT_SOURCE_DIR}/../../services/transport/rust-net/target/${RUST_TRIPLE}/release/libtransport_net.a)
target_link_libraries(transportnative transport_rust android log dl)
